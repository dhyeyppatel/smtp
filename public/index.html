JavaScript (fetch) — Configure Custom SMTP (BYOS) 
// Use your own Gmail/SMTP to avoid loading the platform owner's account
await fetch("https://your-service.vercel.app/api/project/smtp", {
  method: "POST",
  headers: {
    "Content-Type": "application/json",
    "X-API-Key": "sk_abc_dev123"
  },
  body: JSON.stringify({
    user: "my_project@gmail.com",
    password: "my_16_char_app_password",
    reply_to: "support@myproject.com",
    from_name: "My Project Team"
  })
});
JavaScript (fetch) — Enqueue Safe Broadcast (Low Priority) 
const res = await fetch("https://your-service.vercel.app/api/broadcast", {
  method: "POST",
  headers: {
    "Content-Type": "application/json",
    "X-API-Key": "sk_abc_dev123"
  },
  body: JSON.stringify({
    subject: "Monthly Newsletter",
    title: "What's new in our platform",
    message: "<p>Here are your latest product updates...</p>"
  })
});
const data = await res.json();
console.log("Queued Job ID:", data.jobId);

    headers={"X-API-Key": "sk_abc_dev123"},
    json={"email": "user@example.com"}
)
# Returns immediately: {"priority": "immediate", "success": True}
